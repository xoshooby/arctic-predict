            if (!session.active) return;

            if (session.mode === 'standard' && session.round >= session.maxRounds) {
                finishAssessment();
                return;

            session.awaitingChoice = true;
            session.awaitingConfidence = false;
            session.pendingChoice = null;

            // Update Round Counter
            const counterEl = document.getElementById('game-round-counter');
            if (session.mode === 'standard') {
                counterEl.textContent = `Round ${session.round} / 50`;
            } else {
                counterEl.textContent = `Question #${session.round}`;
            }

            // Hide overlays
            document.getElementById('confidence-modal-overlay').classList.add('hidden');
            document.getElementById('outcome-overlay').classList.add('hidden');

            // Select one of the 5 fixed symbol pairs
            const randomIndex = Math.floor(Math.random() * SYMBOL_PAIRS.length);
            session.currentPair = SYMBOL_PAIRS[randomIndex];

            // Render symbols
            document.getElementById('symbol-left-slot').innerHTML = session.currentPair.leftSvg;
            document.getElementById('symbol-right-slot').innerHTML = session.currentPair.rightSvg;

            // Start 3.0s turn countdown
            startTurnTimer();

        function startTurnTimer() {
            clearInterval(session.turnTimerId);
            session.roundStartTime = performance.now();

            const bar = document.getElementById('game-timer-bar');
            const text = document.getElementById('game-turn-timer-text');
            bar.style.width = '100%';
            bar.style.backgroundColor = 'var(--accent-color)';
            text.textContent = '3.0s';

            session.turnTimerId = setInterval(() => {
                if (!session.awaitingChoice) {
                    clearInterval(session.turnTimerId);
                    return;
                }
                const elapsed = performance.now() - session.roundStartTime;
                const remaining = Math.max(0, TURN_LIMIT_MS - elapsed);
                const pct = (remaining / TURN_LIMIT_MS) * 100;

                bar.style.width = `${pct}%`;
                text.textContent = `${(remaining / 1000).toFixed(1)}s`;

                if (remaining < 1000) {
                    bar.style.backgroundColor = '#f43f5e';
                }

                if (remaining <= 0) {
                    clearInterval(session.turnTimerId);
                    handleTurnTimeout();
                }
            }, 30);
        }

        function handlePlayerChoice(choice) {
            if (!session.active || !session.awaitingChoice) return;

            session.awaitingChoice = false;
            clearInterval(session.turnTimerId);

            const reactionMs = Math.min(TURN_LIMIT_MS, Math.round(performance.now() - session.roundStartTime));
            session.pendingChoice = choice;
            session.pendingReactionMs = reactionMs;

            // Check if Standard Mode Rounds 43-50 confidence slider should trigger
            if (session.mode === 'standard' && session.round >= 43 && session.round <= 50) {
                session.awaitingConfidence = true;
                const slider = document.getElementById('conf-slider-input');
                slider.value = 5;
                document.getElementById('conf-slider-val').textContent = '5';
                document.getElementById('confidence-modal-overlay').classList.remove('hidden');
            } else {
                resolveRoundOutcome(choice, reactionMs, null);
            }
        }

        function submitConfidenceAndReveal() {
            if (!session.awaitingConfidence) return;
            session.awaitingConfidence = false;
            const confVal = parseInt(document.getElementById('conf-slider-input').value, 10) || 5;
            document.getElementById('confidence-modal-overlay').classList.add('hidden');
            resolveRoundOutcome(session.pendingChoice, session.pendingReactionMs, confVal);
        }

        function resolveRoundOutcome(choice, reactionMs, confidenceVal) {
            const pair = session.currentPair;
            const roll = Math.random();
            const actualOutcome = roll < pair._pDay ? 'Day' : 'Night';

            const isTimeout = choice === null;
            const isCorrect = !isTimeout && choice === actualOutcome;
            const isOptimal = !isTimeout && choice === pair._fav;

            // Record trial
            session.trials.push({
                round: session.round,
                pairId: pair.id,
                choice: choice,
                actual: actualOutcome,
                isCorrect: isCorrect,
                isOptimal: isOptimal,
                reactionMs: isTimeout ? null : reactionMs,
                confidence: confidenceVal
            });

            // Show immediate outcome overlay for 950ms
            const overlay = document.getElementById('outcome-overlay');
            const iconBox = document.getElementById('outcome-icon-box');
            const actualLabel = document.getElementById('outcome-actual-label');
            const statusTitle = document.getElementById('outcome-status-title');
            const rtSub = document.getElementById('outcome-rt-sub');

            if (actualOutcome === 'Day') {
                iconBox.className = 'w-20 h-20 rounded-full flex items-center justify-center mb-3 text-4xl shadow-lg bg-amber-400 text-slate-950';
                iconBox.innerHTML = '☀️';
                actualLabel.textContent = 'Actual State: DAY';

                iconBox.innerHTML = '🌙';
                actualLabel.textContent = 'Actual State: NIGHT';
            }

            if (isTimeout) {
                statusTitle.textContent = 'TIME EXPIRED';
                statusTitle.className = 'text-3xl sm:text-4xl font-extrabold tracking-tight text-amber-400';
                rtSub.textContent = 'No selection locked within 3.0s';
            } else if (isCorrect) {
                statusTitle.textContent = 'CORRECT PREDICTION';
                statusTitle.className = 'text-3xl sm:text-4xl font-extrabold tracking-tight text-emerald-400';
                rtSub.textContent = `Decision Speed: ${reactionMs} ms`;
            } else {
                statusTitle.textContent = 'INCORRECT';
                statusTitle.className = 'text-3xl sm:text-4xl font-extrabold tracking-tight text-rose-500';
                rtSub.textContent = `Decision Speed: ${reactionMs} ms`;
            }

            overlay.classList.remove('hidden');

            setTimeout(() => {
                if (!session.active) return;
                nextRound();
            }, 950);
        }

        /* =========================================================
           8. END-OF-RUN ANALYTICS & METRICS ENGINE
              (ZERO SPOILERS OF UNDERLYING SYMBOL PROBABILITIES)
           ========================================================= */
        function finishAssessment() {
            stopAllTimers();
            session.active = false;

            const totalQuestions = session.trials.length;
            if (totalQuestions === 0) {
                navigateTo('landing');
                return;
            }

            const totalDurationSec = Math.max(1, Math.round((Date.now() - session.sessionStartTime) / 1000));
            const answeredTrials = session.trials.filter(t => t.choice !== null);
            const timeoutsCount = totalQuestions - answeredTrials.length;

            const correctCount = session.trials.filter(t => t.isCorrect).length;
            const rawAccuracy = Math.round((correctCount / totalQuestions) * 100);

            // Strategy Consistency %: % of answered trials where player chose the mathematically optimal side
            // (after seeing a pair at least once, or overall answered trials)
            const optimalCount = answeredTrials.filter(t => t.isOptimal).length;
            const consistencyPct = answeredTrials.length > 0
                ? Math.round((optimalCount / answeredTrials.length) * 100)
                : 0;

            // Resilience to Variance Score:
            // Whenever a player picks the optimal choice on Pair X, but gets an unlucky roll (isOptimal && !isCorrect),
            // check the VERY NEXT time Pair X appears. Did they stay disciplined with the optimal choice?
            let varianceTests = 0;
            let varianceHeld = 0;
            SYMBOL_PAIRS.forEach(pair => {
                const pairTrials = answeredTrials.filter(t => t.pairId === pair.id);
                for (let i = 0; i < pairTrials.length - 1; i++) {
                    if (pairTrials[i].isOptimal && !pairTrials[i].isCorrect) {
                        varianceTests++;
                        if (pairTrials[i + 1].isOptimal) {
                            varianceHeld++;
                        }
                    }
                }
            });
            const resiliencePct = varianceTests > 0
                ? Math.round((varianceHeld / varianceTests) * 100)
                : consistencyPct;

            // Average Reaction Time
            const avgReactionMs = answeredTrials.length > 0
                ? Math.round(answeredTrials.reduce((acc, t) => acc + t.reactionMs, 0) / answeredTrials.length)
                : 0;

            // Confidence Calibration Score (Standard Mode Rounds 43-50)
            let calibrationPct = null;
            const confTrials = answeredTrials.filter(t => t.confidence !== null);
            if (confTrials.length > 0) {
                // Compare user's 1-10 confidence against pair stability without revealing it
                let totalDiff = 0;
                confTrials.forEach(t => {
                    const pairObj = SYMBOL_PAIRS.find(p => p.id === t.pairId);
                    const strength = Math.max(pairObj._pDay, 1 - pairObj._pDay); // 0.68 to 0.85
                    // Map strength (0.65..0.85) to ideal confidence (6..9)
                    const idealConf = t.isOptimal ? Math.round((strength - 0.5) * 22) : 4;
                    totalDiff += Math.abs(t.confidence - idealConf);
                });
                const avgDiff = totalDiff / confTrials.length;
                calibrationPct = Math.max(0, Math.min(100, Math.round(100 - avgDiff * 11)));
            }

            // Save run to history
            const runRecord = {
                id: Date.now(),
                timestamp: new Date().toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }),
                mode: session.mode,
                questions: totalQuestions,
                correct: correctCount,
                accuracy: rawAccuracy,
                consistency: consistencyPct,
                resilience: resiliencePct,
                avgReactionMs: avgReactionMs,
                timeouts: timeoutsCount,
                durationSec: totalDurationSec,
                calibration: calibrationPct
            };

            appState.history.unshift(runRecord);
            saveHistoryToStorage();

            // Render Results Dashboard
            populateResultsDashboard(runRecord);
            navigateTo('results');
        }

                    <td class="py-3.5 font-mono-num text-xs sm:text-sm font-bold">${count > 0 ? winRate + '%' : '—'}</td>
                    <td class="py-3.5 font-mono-num text-xs sm:text-sm text-right">${avgRt}</td>
                `;

            });
        }

            document.getElementById('std-best-cons').textContent = stdRuns.length > 0 ? `${Math.max(...stdRuns.map(r => r.consistency))}%` : '—';

                            <td class="py-2.5 font-mono-num text-emerald-500 font-bold">${r.consistency}%</td>
                            <td class="py-2.5 font-mono-num text-right">${r.avgReactionMs} ms</td>

            document.getElementById('end-best-cons').textContent = endRuns.length > 0 ? `${Math.max(...endRuns.map(r => r.consistency))}%` : '—';

                ? `${Math.round(endRuns.reduce((a, b) => a + b.avgReactionMs, 0) / endRuns.length)}ms`
                : '—';

            // Endurance Recent 3 Table (Includes Questions Answered & Duration)
            const endTbody = document.getElementById('end-recent-tbody');

                endTbody.innerHTML = `<tr><td colspan="5" class="py-4 text-center theme-text-muted">No Endurance Free Play runs recorded yet.</td></tr>`;

                endRuns.slice(0, 3).forEach(r => {
                    endTbody.innerHTML += `
                        <tr>
                            <td class="py-2.5">${r.timestamp}</td>
                            <td class="py-2.5 font-mono-num font-semibold">${r.questions} Qs</td>
                            <td class="py-2.5 font-mono-num font-bold">${r.accuracy}%</td>
                            <td class="py-2.5 font-mono-num text-emerald-500 font-bold">${r.consistency}%</td>
                            <td class="py-2.5 font-mono-num text-right">${formatDuration(r.durationSec)}</td>
                        </tr>
                    `;
                });
            }
        }

        function setHistoryFilter(filter) {
            appState.historyFilter = filter;
            ['all', 'standard', 'endurance'].forEach(tab => {
                const btn = document.getElementById(`hist-tab-${tab}`);
                if (tab === filter) {
                    btn.className = 'theme-accent-btn px-3.5 py-2 rounded-xl text-xs font-bold';
                } else {
                    btn.className = 'theme-card px-3.5 py-2 rounded-xl text-xs font-bold';
                }
            });
            renderFullHistoryTable();
        }

        function renderFullHistoryTable() {
            const tbody = document.getElementById('full-history-tbody');
            tbody.innerHTML = '';

            let filtered = appState.history;
            if (appState.historyFilter !== 'all') {
                filtered = appState.history.filter(r => r.mode === appState.historyFilter);
            }

            const recent15 = filtered.slice(0, 15);
            if (recent15.length === 0) {
                tbody.innerHTML = `<tr><td colspan="9" class="py-8 text-center theme-text-muted">No sessions recorded for this filter yet. Launch a game mode to start tracking!</td></tr>`;
                return;
            }

            recent15.forEach((r, idx) => {
                const modeBadge = r.mode === 'standard'
                    ? `<span class="px-2.5 py-0.5 rounded-full text-[11px] font-bold bg-blue-500/15 text-blue-400">Standard (50)</span>`
                    : `<span class="px-2.5 py-0.5 rounded-full text-[11px] font-bold bg-amber-500/15 text-amber-400">Endurance</span>`;

                const extraMetric = r.mode === 'standard'
                    ? `Calib: ${r.calibration !== null ? r.calibration + '%' : '—'}`
                    : `Time: ${formatDuration(r.durationSec)}`;

                tbody.innerHTML += `
                    <tr>
                        <td class="py-3 font-mono-num text-xs theme-text-muted">#${idx + 1}</td>
                        <td class="py-3 text-xs sm:text-sm">${r.timestamp}</td>
                        <td class="py-3">${modeBadge}</td>
                        <td class="py-3 font-mono-num font-semibold">${r.questions} Qs</td>
                        <td class="py-3 font-mono-num font-bold">${r.accuracy}%</td>
                        <td class="py-3 font-mono-num font-bold text-emerald-500">${r.consistency}%</td>
                        <td class="py-3 font-mono-num font-bold text-sky-400">${r.resilience}%</td>
                        <td class="py-3 font-mono-num">${r.avgReactionMs} ms</td>
                        <td class="py-3 font-mono-num text-xs text-right">${extraMetric}</td>
                    </tr>
                `;
            });
        }

        function formatDuration(totalSec) {
            const mins = Math.floor(totalSec / 60);
            const secs = totalSec % 60;
            return `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
        }
